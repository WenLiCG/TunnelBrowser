cmake_minimum_required(VERSION 3.21)
project(TunnelBrowser LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/runtime.json" runtime)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/runtime.json")
string(JSON CEF_VERSION GET "${runtime}" cef)
string(JSON XRAY_VERSION GET "${runtime}" libxray)
set(CEF_ROOT "${CMAKE_SOURCE_DIR}/third_party/cef_binary_${CEF_VERSION}_windows64_minimal")
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CEF_ROOT}/cmake")
set(PROJECT_ARCH "x86_64")
set(USE_SANDBOX ON CACHE BOOL "Keep Chromium sandbox enabled" FORCE)
set(USE_ATL OFF CACHE BOOL "No ATL dependency" FORCE)
find_package(CEF REQUIRED)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
add_library(TunnelBrowser SHARED src/app.cc src/node.cc src/profile.cc)
SET_LIBRARY_TARGET_PROPERTIES(TunnelBrowser)
target_include_directories(TunnelBrowser PRIVATE "${CEF_ROOT}" "${CMAKE_SOURCE_DIR}/src")
target_link_libraries(TunnelBrowser PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
option(TB_BUILD_TESTS "Build a separate browser DLL with local test entry points" OFF)
if(TB_BUILD_TESTS)
  add_library(TunnelBrowserTest SHARED src/app.cc src/node.cc src/profile.cc)
  SET_LIBRARY_TARGET_PROPERTIES(TunnelBrowserTest)
  target_include_directories(TunnelBrowserTest PRIVATE "${CEF_ROOT}" "${CMAKE_SOURCE_DIR}/src")
  target_link_libraries(TunnelBrowserTest PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  target_compile_definitions(TunnelBrowserTest PRIVATE TB_TESTING XRAY_VERSION="${XRAY_VERSION}")
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  # CEF diagnostics embed __FILE__ in the release DLL.
  cmake_path(NATIVE_PATH CMAKE_SOURCE_DIR NORMALIZE build_source_path)
  foreach(target TunnelBrowser libcef_dll_wrapper)
    target_compile_options(${target} PRIVATE
      "$<$<CONFIG:Release>:/experimental:deterministic>"
      "$<$<CONFIG:Release>:/pathmap:${build_source_path}=.>")
  endforeach()
  target_compile_options(TunnelBrowser PRIVATE "$<$<CONFIG:Release>:/Brepro>")
  target_link_options(TunnelBrowser PRIVATE "$<$<CONFIG:Release>:/Brepro>")
endif()
function(stage_browser target directory)
  set(run "${CMAKE_BINARY_DIR}/${directory}")
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${run}" OUTPUT_NAME TunnelBrowser
    ARCHIVE_OUTPUT_DIRECTORY "${run}")
  string(JSON count LENGTH "${runtime}" cef_files)
  math(EXPR last "${count} - 1")
  foreach(index RANGE ${last})
    string(JSON file_path GET "${runtime}" cef_files ${index})
    string(FIND "${file_path}" "/" separator)
    math(EXPR subpath "${separator} + 1")
    string(SUBSTRING "${file_path}" ${subpath} -1 file_name)
    configure_file("${CEF_ROOT}/${file_path}" "${run}/${file_name}" COPYONLY)
  endforeach()
  configure_file("${CEF_ROOT}/Release/bootstrap.exe" "${run}/TunnelBrowser.exe" COPYONLY)
  configure_file("${PROJECT_SOURCE_DIR}/third_party/libxray/libxray-windows-x64/libXray.dll"
    "${run}/libXray.dll" COPYONLY)
endfunction()

stage_browser(TunnelBrowser app)
if(TB_BUILD_TESTS)
  stage_browser(TunnelBrowserTest test)
  configure_file("${CEF_ROOT}/Release/bootstrapc.exe" "${CMAKE_BINARY_DIR}/test/bootstrapc.exe" COPYONLY)
endif()
